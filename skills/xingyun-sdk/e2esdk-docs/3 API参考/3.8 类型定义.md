# 3.8 类型定义

本节汇总 `XingyunAvatarAgent` 涉及的枚举与接口。

## 状态枚举

### `AgentState`

```ts
type AgentState =
  | "idle" | "initializing" | "ready" | "running"
  | "reconnecting" | "stopped" | "destroyed" | "failed";
```

### `AgentSocketState`

```ts
type AgentSocketState = "idle" | "connecting" | "open" | "closed" | "error";
```

### `AgentASRState`

```ts
type AgentASRState =
  | "idle" | "requesting-permission" | "starting" | "listening" | "stopping" | "failed";
```

### `AgentConversationState`

```ts
type AgentConversationState =
  | "idle" | "asking" | "speaking-directly" | "speaking"
  | "completed" | "interrupted" | "failed";
```

### `AgentErrorDomain`

```ts
type AgentErrorDomain = "permission" | "network" | "asr" | "brain" | "ttsa" | "quota" | "sdk";
```

### `AvatarStatus`（数值枚举）

```ts
enum AvatarStatus {
  online = 0,
  offline = 1,
  close = 4,
  visible = 5,
  invisible = 6,
}
```

### `RenderState`

```ts
type RenderState = "init" | "rendering" | "pausing" | "paused" | "resumed" | "stopped";
```

### `InitModel`

```ts
type InitModel = "normal" | "invisible";
```

## Agent 结构体

### `AgentAudioOptions`

```ts
interface AgentAudioOptions {
  inputEnabled?: boolean; // 初始化后是否立即开启音频输入
}
```

### `AgentReconnectOptions`

```ts
interface AgentReconnectOptions {
  enabled?: boolean;       // 是否开启自动重连
  maxAttempts?: number;    // 短重试最大次数
  initialDelayMs?: number; // 初次重试延迟（毫秒）
  maxDelayMs?: number;     // 重试延迟上限（毫秒）
}
```

### `AgentError`

```ts
interface AgentError {
  code: string;
  message: string;
  domain: AgentErrorDomain;
  retryable?: boolean;
  cause?: unknown;
}
```

### `AgentASRResult`

```ts
interface AgentASRResult {
  text: string;
  isFinal: boolean;
  raw?: unknown;
}
```

### `AgentLLMUsage`

```ts
interface AgentLLMUsage {
  promptTokens: number;
  completionTokens: number;
  totalTokens: number;
  cachedTokens: number;
}
```

### `AgentLLMResponse`

```ts
interface AgentLLMResponse {
  event: "chunk" | "done";
  text?: string;
  isFirst?: boolean;
  usage?: AgentLLMUsage;
  raw?: unknown;
}
```

### `AgentSemanticJudgeResult`

```ts
interface AgentSemanticJudgeResult {
  query: string;
  meaningful: boolean;
  action: string;
  raw?: unknown;
}
```

### `AgentSpeakStateEvent`

```ts
interface AgentSpeakStateEvent {
  state: string;
  clientSpeakId?: string | number;
  raw?: unknown;
}
```

### `AgentConversationEvent`

```ts
interface AgentConversationEvent {
  state: AgentConversationState;
  text?: string;
  raw?: unknown;
}
```

### `AgentServiceConfig`

```ts
type AgentServiceConfig = Record<string, unknown>;
```

### `AgentBrainConfig`

```ts
interface AgentBrainConfig extends AgentServiceConfig {
  extra_body?: Record<string, unknown>; // 仅 OpenAI-compatible 后端下发
}
```

### `AgentFeatures`

```ts
interface AgentFeatures extends AgentServiceConfig {
  anti_interference?: {
    semantic_judge_enabled?: boolean; // 语义判干扰（防播报被闲聊误触发）
  };
  vad_merge_mode?: boolean;
  volume_and_repetition_text_detection?: boolean;
}
```

### `XingyunAvatarAgentCallbacks`

```ts
interface XingyunAvatarAgentCallbacks {
  onAgentStateChange?: (state: AgentState) => void;
  onSocketStateChange?: (state: AgentSocketState) => void;
  onASRStateChange?: (state: AgentASRState) => void;
  onASRResult?: (result: AgentASRResult) => void;
  onLLMResponse?: (response: AgentLLMResponse) => void;
  onSpeakStateChange?: (event: AgentSpeakStateEvent) => void;
  onConversationChange?: (event: AgentConversationEvent) => void;
  onSemanticJudgeResult?: (result: AgentSemanticJudgeResult) => void;
  onRenderChange?: (state: RenderState) => void;
  onNetworkInfo?: (info: INetworkInfo) => void;
  onError?: (error: AgentError) => void;
}
```

### `XingyunAvatarAgentInitParams`

```ts
type XingyunAvatarAgentInitParams = {
  onDownloadProgress?: (progress: number) => void;
  initModel?: InitModel;
};
```

### `IAvatarOptions`（公开子集）

```ts
interface IAvatarOptions {
  appId: string;
  appSecret: string;
  gatewayServer: string;
  onMessage: (error: SDKError) => void;
  containerId?: string;
  container?: HTMLElement;
  config?: AvatarConfig;
  enableClientInterrupt?: boolean; // 默认 true
  enableDebugger?: boolean;        // 默认 false
  onStatusChange?: (status: AvatarStatus) => void;
  onRenderChange?: (state: RenderState) => void;
  onVoiceStateChange?: (state: string, duration?: number, client_speak_id?: string | number) => void;
  onSpeakStateChange?: (state: string, client_speak_id: string | number) => void;
  onWalkStateChange?: (state: string) => void;
  onWidgetEvent?: (data: IRawWidgetData) => void;
  proxyWidget?: { [type: string]: (data: any) => void };
  onNetworkInfo?: (info: INetworkInfo) => void;
  onStartSessionWarning?: (message: object) => void;
  onFPSUpdate?: (stats: FPSStats) => void;
}
```

## 具身智能体渲染结构体

### `SDKError`

```ts
interface SDKError {
  code: number;
  message: string;
  timestamp: number;
  originalError?: unknown;
}
```

### `AvatarConfig`

```ts
interface AvatarConfig {
  layout?: Layout;
  walk_config?: WalkConfig;
  background_img?: string;
  raw_audio?: boolean;
  init_events?: unknown[];
  framedata_proto_version?: string;
  resolution?: unknown;
}
```

### `Layout`

```ts
interface Layout {
  container: {
    size: number[]; // 容器尺寸 [width, height]
  };
  avatar: {
    v_align: string;      // 垂直对齐：'top' | 'center' | 'bottom'
    h_align: string;      // 水平对齐：'left' | 'center' | 'right'
    scale: number | string; // 缩放：数字倍率或 vh 单位
    offset_x: number;     // 水平偏移（像素）
    offset_y: number;     // 垂直偏移（像素）
  };
}
```

### `WalkConfig`

```ts
interface WalkConfig {
  min_x_offset: number;
  max_x_offset: number;
  walk_points: { [key: string]: number }; // 停靠点：名称 → x 坐标
  init_point?: number;                    // 初始停靠点
}
```

### `INetworkInfo`

```ts
interface INetworkInfo {
  rtt: number;            // 打包往返时延
  downlink: number;       // 下行带宽（Mbps）
  networkStatus: boolean; // 网络状态
}
```

### `FPSStats`

```ts
interface FPSStats {
  /* 帧率统计信息 */
  [key: string]: unknown;
}
```

### `IRawWidgetData`

```ts
interface IRawWidgetData {
  type: string;
  [key: string]: unknown;
}
```
